# Below we provide the steps for:

# 1: Installing in an empty VM with ubuntu 22.04 (it should work with other versions)
# 2: Installing using a vagrant file that creates the same VM
# 3: The configuration of Metabase needed in both cases


##### 1: Installation steps for VM #####

# This is just to be able to copy and paste into the VM
sudo apt update
sudo apt install build-essential dkms linux-headers-$(uname -r)
sudo mount /dev/cdrom /mnt
sudo /mnt/VBoxLinuxAdditions.run
# turnoff the VM, activate VBox options to copy and paste


sudo apt-get update

sudo apt-get install -y \
 software-properties-common \
 build-essential \
 zlib1g-dev \
 libssl-dev \
 libffi-dev \
 curl \
 libbz2-dev \
 libsqlite3-dev \
 liblzma-dev \
 mysql-server \
 libmysqlclient-dev \
 pkg-config \
 openjdk-11-jdk


sudo systemctl start mysql
sudo systemctl enable mysql

sudo mysql -e "CREATE DATABASE edge_computing;"
sudo mysql -e "CREATE USER 'admin'@'%' IDENTIFIED BY 'admin';"
sudo mysql -e "GRANT ALL PRIVILEGES ON edge_computing.* TO 'admin'@'%';"
sudo mysql -e "FLUSH PRIVILEGES;"




# Compile and install python 3.12

if ! command -v python3.12 &> /dev/null; then
  cd /tmp
  curl -O https://www.python.org/ftp/python/3.12.1/Python-3.12.1.tgz
  tar -xf Python-3.12.1.tgz
  cd Python-3.12.1
  ./configure --enable-optimizations
  make -j "$(nproc)"
  sudo make altinstall
  cd /tmp
  rm -rf Python-3.12.1 Python-3.12.1.tgz
fi

curl -sS https://bootstrap.pypa.io/get-pip.py | sudo python3.12

# Download code
sudo apt install git
# To install in home folder
cd ~
git clone https://github.com/Santiago-Tabarez/edge-computing-simulations
cd edge-computing-simulations

# Activate virtual environment
python3.12 -m venv ~/env
source ~/env/bin/activate

# Download project dependencies
pip install --upgrade pip
pip install -r requirements.txt


# Execute the simulator
# Replace "edge-user" with real username
export PYTHONPATH=/home/edge-user/edge-computing-simulations
python3 main/main.py

curl -L -o ~/metabase.jar https://downloads.metabase.com/v0.44.6/metabase.jar
nohup java -jar ~/metabase.jar &


##### 2: Installation steps for Vagrant #####

# Install Vagrant and virtual Box, in a windows or unix terminal run

vagrant up

vagrant ssh

source /home/vagrant/env/bin/activate

cd /home/vagrant/project

python3 main/main.py
